    前言

    在《数据结构与算法c/c++版》上抄程序的时候发现了一些笔误，所以有了这个系列的文章，这里的程序是自己验证过可运行的，也是一个学习的记录 (^ - ^)

    程序1.1

    抽象数据类型

    #include <iostream>

/*------------------------------------*/

static double pi = 3.14; 

struct Circle {
    double radius;
};

double Area(struct Circle *c) {
    return pi * c->radius * c->radius;
}

double Perimeter(struct Circle *c) {
    return 2 * pi * c->radius;
}

/*------------------------------------*/

int main() {
    Circle *circle = new Circle();
    circle->radius = 3.0;
    double area = Area(circle);
    double perimeter = Perimeter(circle);
    std::cout << "Area: " << area << std::endl;
    std::cout << "Perimeter:" << perimeter << std::endl;
    
	delete circle;
    return 0;
}

    程序1.2-1.3

    抽象数据类型

    #include <iostream>
#include <vector> 
#include <memory>

/*------------------------------------*/

struct Shape {
	static double pi;
	virtual double Area() = 0;
	virtual double Perimeter() = 0;
	virtual ~Shape() = default;
};

double Shape::pi = 3.14;

/*------------------------------------*/

// 圆形 
struct Circle : public Shape {
	double _radius = 0;
	
	Circle(double r) : _radius(r) {}
	
	double Area() override {
		return pi * _radius * _radius;
	}
	
	double Perimeter() override {
		return 2 * pi * _radius;
	}
};

/*------------------------------------*/

// 长方形 
struct Rectangle : public Shape {
	double _length, _width;
	
	Rectangle(double l, double w) : _length(l), _width(w) {}
	
	double Area() override {
		return _length * _width;
	}
	
	double Perimeter() override {
		return 2 * (_length + _width);
	}
};


int main() {
	/* 法一 
	Shape *s1 = new Circle(3.0);
	
	std::cout << "Area: " << s1->Area() << std::endl;
	std::cout << "Perimeter: " << s1->Perimeter() << std::endl;
	
	Shape *s2 = new Rectangle(2.0, 3.0);
	
	std::cout << "Area: " << s2->Area() << std::endl;
	std::cout << "Perimeter: " << s2->Perimeter() << std::endl;
	
	delete s1;
	delete s2;
	*/ 
	
	/* 法二 
	Shape *shapes[2];
	shapes[0] = new Circle(3.0);
	shapes[1] = new Rectangle(2.0, 3.0);
	
	for (int i = 0; i < 2; i++) {
		std::cout << "Area: " << shapes[i]->Area() << std::endl;
		std::cout << "Perimeter: " << shapes[i]->Perimeter() << std::endl; 
		delete shapes[i];
	}
	*/
	
	/* 法三 （智能指针） 
	std::vector<std::unique_ptr<Shape>> shapes;
	shapes.push_back(std::make_unique<Circle>(3.0));
	shapes.push_back(std::make_unique<Rectangle>(2.0, 3.0));
	
	for (auto &s : shapes) {
		std::cout << "Area: " << s->Area() << std::endl;
		std::cout << "Perimeter: " << s->Perimeter() << std::endl; 
	}
	*/ 
	
	std::vector<Shape*> shapes;
	shapes.push_back(new Circle(3.0));
	shapes.push_back(new Rectangle(2.0, 3.0));
	
	for (auto *s : shapes) {
		std::cout << "Area: " << s->Area() << std::endl;
		std::cout << "Perimeter: " << s->Perimeter() << std::endl; 
		delete s;
	}
	 
	return 0;
}

    程序1.4

    欧几里得算法

    #include <iostream>

int gcd(int a, int b) {
	int r = a % b;
	while (r != 0) {
		a = b;
		b = r;
		r = a % b;
	}
	return b;
} 

/* 递归版
int gcd(int a, int b) {
	return b == 0 ? a : gcd(b, a % b);
}
*/

int main() {
	int a = 136;
	int b = 438;
	int g = gcd(a, b);
	std::cout << "gcd(a, b) = " << g << std::endl;
	
	return 0;
}
