		return -1;
	}
	
	//保存 
	ElementType removed = list->data[idx];
	
	// 挪（删）
	for (int i = idx; i < list->length - 1; i++) {      // length-1 防止越界  
		list->data[i] = list->data[i+1];
	} 
	list->length--;
	return removed; 
}

		if (list->data[idx] == key) return idx;
	}

    seqlist.c

    #include "seqlist.h"
#include <stdlib.h>
#include <assert.h>

void initialize(LList *list) {
    assert (list != NULL);
    
    list ->_capacity = 32;
    list ->_data = (ElementType*) malloc(list->_capacity * sizeof(ElementType));
    
    if (list->_data == NULL) {
        fprintf(stderr, "not enough memory! \n");
        exit(1);
    }
    list->_length = 0;
}

void destroy(LList *list) {
    assert(list != NULL);
    free(list->_data);
    list->_data = NULL;
    list->_length = 0;
    list->_capacity = 0;
}

void clear_all_data(LList *list) {
    assert(list != NULL);
    //if (list->_length == 0) return;
    list->_length = 0;
}

bool is_empty(LList *list) {
    assert(list != NULL); 
    return list->_length == 0;
}

size_t size(LList *list) {
    assert(list != NULL);
    return list->_length;
}

ElementType get(LList *list, int idx) {
    assert(list != NULL && idx >= 0 && idx <list->_length);
    return list->_data[idx];
}

void set(LList *list, int idx, ElementType d) {
    assert(list != NULL && idx >= 0 && idx <list->_length);
    list->_data[idx] = d;
}

void add(LList *list, int idx, ElementType d) {
    assert(list != NULL && idx >= 0 && idx <= list->_length);
    
    ElementType *temp = list->_data;

    if (list->_capacity < list->_length + 1) {
        list->_capacity *= 2;
        temp = (ElementType*) malloc(sizeof(ElementType) * list->_capacity);
        if (temp == NULL) {
            fprintf(stderr, "not enough memory! \n");
            exit(1);
        }
        for (size_t i = 0; i < idx; ++i) {
            temp[i] = list->_data[i];
        }
    }

    for (size_t i = list->_length; i > idx; --i) {
        temp[i] = list->_data[i - 1];
    }
    temp[idx] = d;

    if (temp != list->_data) {
        free(list->_data);
        list->_data = temp;
    }
    ++list->_length;
}

ElementType remove_node(LList *list, int idx) {
    assert(list != NULL && idx >= 0 && idx < list->_length);
    ElementType res = list->_data[idx];
    for (int i = idx; i < list->_length - 1; ++i) {
        list->_data[i] = list->_data[i + 1];
    }
    --list->_length;
    return res;
}

int index_of(LList *list, ElementType key) {
    for (int i = 0; i < list->_length; ++i) {
        if (list->_data[i] == key) {
            return i;
        }
    }
    return -1;
}

    linear_list_cpp.h

    #ifndef CHAPTER_2_LINEAR_LIST_CPP_H
#define CHAPTER_2_LINEAR_LIST_CPP_H

#include <cassert>
#include <cstddef>

template<typename ElementType> 
class Linear_List {
protected:
    size_t _length = 0;
public:
    Linear_List(){}
    ~Linear_List(){}

    size_t size() const {
        return _length;
    }
    bool is_empty() const {
        return _length == 0;
    }

    virtual void clear_all_data() = 0;
    virtual ElementType get(size_t idx) const = 0;
    virtual void set(size_t idx, ElementType d) = 0;
    virtual void add(size_t idx, ElementType d) = 0;
    virtual ElementType remove_node(size_t idx) = 0;
    virtual int index_of(ElementType key) const = 0;
};


template<typename ElementType>
class SeqList : public Linear_List<ElementType> {
private:
    ElementType *_data = nullptr;
    size_t _capacity = 0;
public:
    SeqList(size_t size = 32) {
        _data = new ElementType[size];
        _capacity = size;
    }
    ~SeqList() {
        delete [] _data;
    }

    void clear_all_data() {
        this->_length = 0;
    }

    ElementType get(size_t idx) const {
        assert(idx < this->_length);
        return _data[idx];
    }

    void set(size_t idx, ElementType d) {
        assert(idx < this->_length);
        _data[idx] = d;
    }

    void add(size_t idx, ElementType d) {
        assert(_data != NULL && idx <= this->_length);
        ElementType *temp = _data;
        if (_capacity < this->_length + 1) {
            _capacity *= 2;
            temp = new ElementType[_capacity];
            /* new完会抛异常，所以我并不理解这个检查的意义,这里决定删掉 (而且用的是 NULL 还不是 nullptr...
            if (temp == NULL) {
                fprintf(stderr, "not enough memory! \n");
                exit(1);
            }
            */
            for(size_t i = 0; i < idx; ++i) {
                temp[i] = _data[i];
            }
        }
        for(size_t i = this->_length; i > idx; --i) {
            temp[i] = _data[i - 1];
        }
        temp[idx] = d;
        if(temp != _data) {
            delete [] _data;
            _data = temp;
        }
        ++this->_length;
    }

    ElementType remove_node(size_t idx) {
        assert(idx < this->_length);
        ElementType removed = _data[idx];
        for (size_t i = idx + 1; i < this->_length; ++i) {
            _data[i - 1] = _data[i];
        }
        --this->_length;
        return removed;
    }

    int index_of(ElementType key) const {
        for (size_t i = 0; i < this->_length; ++i) {
            if (_data[i] == key) return i;
        }
        return -1;
    }
};
#endif

#define LINKED_LIST_H

typedef int ElementType;
typedef struct Linear_List LList;

struct Node {
    ElementType _data;
    struct Node *_next;
};

struct Linear_List {
    size_t _length;
    struct Node *_head;
    struct Node *_tail;
};

void initialize(LList *list);
void destroy(LList *list);

bool is_empty(LList *list);
size_t size(LList *list);
ElementType get(LList *list, size_t idx);

void add(LList *list, size_t idx, ElementType d);
ElementType remove_node(LList *list, size_t idx);
int index_of(LList *list, ElementType key);
void push_front(LList *list, ElementType d);
void push_back(LList *list, ElementType d);
ElementType pop_front(LList *list);
ElementType pop_back(LList *list);

#endif

    linked_list.c

    #include <stdlib.h>
#include <assert.h>
#include "linked_list.h"

void initialize(LList *list) {
    list->_head = NULL;
    list->_tail = NULL;
    list->_length = 0;
}

void destroy(LList *list) {
    clear_all_data(list);
}

//头指针往后指，指一个删一个，直到删完
void clear_all_data(LList *list) {
    assert(list != NULL);
    if (list->_head == NULL) return;
    struct Node *t = list->_head;
    while (t != NULL) {
        list->_head = t->_next;
        free(t);
        t = list->_head;
    }
    list->_head = NULL;
    list->_tail = NULL;
    list->_length = 0;
}

bool is_empty(LList *list) {
    assert(list != NULL);
    return list->_length == 0;
}

size_t size(LList *list) {
    assert(list != NULL);
    return list->_length;
}

struct Node *_get_node(LList *list, size_t idx) {
    assert(list != NULL);
    if (idx >= list->_length) return NULL;
    size_t i = 0;
    struct Node *t = list->_head;
    while (i < idx) {
        t = t->_next;
        ++i;
    }
    return t;
}

ElementType get(LList *list, size_t idx) {
    assert(list != NULL);
    struct Node *n = _get_node(list, idx);
    assert(n != NULL);
    return n->_data;
}

void set(LList *list, size_t idx, ElementType d) {
    assert(list != NULL);
    struct Node *n = _get_node(list, idx);
    assert(n != NULL);
    n->_data = d;
}

void add(LList *list, size_t idx, ElementType d) {
    assert(list != NULL);
    /* 实话说这个特判也没看懂，_length = 0 时 idx 也只能 = 0，那有下面的分支足够了呀
    if (list->_length == 0) {
        push_front(list, d);
        return;
    }
    */
    assert (idx <= list->_length);
    if (idx == 0) {
        push_front(list, d);
    }
    else if (idx == list->_length) {
        push_back(list, d);
    }
    else {
        struct Node *pre = _get_node(list, idx - 1);                 //先找到前一个结点再插
        struct Node *new_node = (struct Node*) malloc(sizeof(struct Node));
        new_node->_data = d;
        new_node->_next = pre->_next;
        pre->_next = new_node;
        ++list->_length;
    }
}

ElementType remove_node(LList *list, size_t idx) {
    assert(list != NULL && idx < list->_length);
    ElementType removed;
    if (idx == 0) {
        removed = list->_head->_data;
        pop_front(list);
    }
    else if (idx == list->_length - 1) {
        removed = list->_tail->_data;
        pop_back(list);
    }
    else {
        struct Node *pre = _get_node(list, idx - 1);
        removed = pre->_next->_data;
        struct Node *cur = pre->_next;
        pre->_next = cur->_next;
        free(cur);
        --list->_length;
    }
    return removed;
}

int index_of(LList *list, ElementType key) {
    assert(list != NULL);
    struct Node *t = list->_head;
    for (size_t i = 0; i < list->_length; ++i, t = t->_next) {
        if (t->_data == key) return i;
    }
    return -1;
}

void push_front(LList *list, ElementType d) {
    assert(list != NULL);
    struct Node *new_node = (struct Node*) malloc(sizeof(struct Node));
    assert(new_node != NULL);
    new_node->_data = d;
    new_node->_next = NULL;
    if (list->_head == NULL) {
        list->_head = list->_tail = new_node;
    }
    else {
        new_node->_next = list->_head;
        list->_head = new_node;
    }
    ++list->_length;
}

void push_back(LList *list, ElementType d) {
    assert(list != NULL);
    struct Node *new_node = (struct Node*) malloc(sizeof(struct Node));
    assert(new_node != NULL);
    new_node->_data = d;
    new_node->_next = NULL;
    if (list->_tail == NULL) {
        list->_head = list->_tail = new_node;
    }
    else {
        list->_tail->_next = new_node;
        list->_tail = new_node;
    }
    ++list->_length;
}

ElementType pop_front(LList *list) {
    assert(list != NULL);
    assert(list->_length > 0);
    //if (list->_length == 0) return -1;
    struct Node *t = list->_head;
    ElementType removed = t->_data;
    if (list->_length == 1) {                //只有一个点则置空
        free(t);
        list->_head = list->_tail = NULL;
    }
    else {
        list->_head = t->_next;
        free(t);
    }
    --list->_length;
    return removed;
}

ElementType pop_back(LList *list) {
    assert(list != NULL);
    assert(list->_length > 0);
    //if (list->_length == 0) return -1;
    struct Node *t = list->_tail;
    ElementType removed = t->_data;
    if (list->_length == 1) {
        free(t);
        list->_head = list->_tail = NULL;
    }
    else {
        struct Node *pre = _get_node(list, list->_length - 2);      //找倒数第二个点
        pre->_next = NULL;
        free(list->_tail);
        list->_tail = pre;
    }
    --list->_length;
    return removed;
}

    print_all(&list);
    printf("\n");

    r = remove_node(&list, size(&list) - 1);
    printf("removed = %d\n", r);
    printf("list = ");
    print_all(&list);
    printf("\n");

    // ==================== 11. pop_front / pop_back ====================
    printf("pop_front = %d\n", pop_front(&list));
    printf("pop_back  = %d\n", pop_back(&list));
    printf("list = ");
    print_all(&list);
    printf("\nsize = %zu\n", size(&list));

    // ==================== 12. index_of ====================
    printf("index_of(2)  = %d\n", index_of(&list, 2));
    printf("index_of(99) = %d\n", index_of(&list, 99));

    // ==================== 13. clear_all_data ====================
    clear_all_data(&list);
    printf("is_empty = %d\n", is_empty(&list));
    printf("size = %zu\n", size(&list));

    // ==================== 14. 清空后还能用吗 ====================
    push_back(&list, 7);
    push_back(&list, 8);
    printf("list = ");
    print_all(&list);
    printf("\nsize = %zu\n", size(&list));

    // ==================== 15. destroy ====================
    destroy(&list);
    printf("destroy finished\n");

    return 0;
}

    #ifndef CHAPTER_2_C_LINEAR_LIST_H
#define CHAPTER_2_C_LINEAR_LIST_H

#include <cassert>
#include <cstddef>

template<typename ElementType>
class Linear_List {
protected:
    size_t _length = 0;
public:
    Linear_List() {}
    ~Linear_List() {}

    size_t size() const {
        return _length;
    }

    bool is_empty() const {
        return _length == 0;
    }

    virtual void clear_all_data() = 0;
    virtual ElementType get(size_t idx) const = 0;
    virtual void set(size_t idx, ElementType d) = 0;
    virtual void add(size_t idx, ElementType d) = 0;
    virtual ElementType remove_node(size_t idx) = 0;
    virtual int index_of(ElementType key) const = 0;
};

template<typename ElementType>
class LinkedList : public Linear_List<ElementType> {
private:
    struct Node {
        ElementType _data;
        Node *_next = nullptr;
    };
    Node *_head = nullptr;
    Node *_tail = nullptr;

    Node *_get_node(size_t idx) const {
        assert(idx < this->_length);
        size_t i = 0;
        Node *t = _head;
        while (i < idx) {
            t = t->_next;
            ++i;
        }   
        return t;
    }
    
public:
    LinkedList() {};
    ~LinkedList() {
        clear_all_data();
    }

    void clear_all_data() {
        while (_head != nullptr) {
            Node *t = _head->_next;
            delete _head;
            _head = t;
        }
        _head = _tail = nullptr;
        this->_length = 0;
    }

    ElementType get(size_t idx) const {
        assert(idx < this->_length);
        Node *t = _get_node(idx);
        return t->_data;
    }

    void set(size_t idx, ElementType d) {
        assert(idx < this->_length);
        Node *t = _get_node(idx);
        t->_data = d;
    }

    void add(size_t idx, ElementType d) {
        assert(idx <= this->_length);
        if (idx == this->_length) {
            push_back(d);
            return;
        }
        if (idx == 0) {
            push_front(d);
            return;
        }
        Node *pre = _get_node(idx - 1);
        Node *now = new Node;
        now->_data = d;
        now->_next = pre->_next;
        pre->_next = now;
        ++this->_length;
    }

    int index_of(ElementType key) const {
        Node *t = _head;
        for (size_t i = 0; i < this->_length; ++i, t = t->_next) {
            if (t->_data == key) return i;
        }
        return -1;
    }

    ElementType remove_node (size_t idx) {
        assert(idx < this->_length);
        if (idx == 0) {
            ElementType removed = _head->_data;
            pop_front();
            return removed;
        }
        if (idx == this->_length - 1) {
            ElementType removed = _tail->_data;
            pop_back();
            return removed;
        }
        Node *pre = _get_node(idx - 1);
        Node *cur = pre->_next;
        pre->_next = cur->_next;
        ElementType removed = cur->_data;
        delete cur;
        --this->_length;
        return removed;
    }

    void push_front(ElementType d) {
        Node *t = new Node;
        t->_data = d;
        if (_head == nullptr) {
            _tail = _head = t;
        }
        else {
            t->_next = _head;
            _head = t;
        }
        ++this->_length;
    }

    void push_back(ElementType d) {
        Node *t = new Node;
        t->_data = d;
        if (_tail == nullptr) {
            _head = _tail = t;
        }
        else {
            _tail->_next = t;
            _tail = t;
        }
        ++this->_length;
    }

    void pop_front() {
        assert(_head != nullptr);
        if (_head == _tail) {
            delete _head;
            _head = _tail = nullptr;
        }
        else {
            Node *t = _head->_next;
            delete _head;
            _head = t;
        }
        --this->_length;
    }

    void pop_back() {
        assert(_head != nullptr);
        if (_head == _tail) {
            delete _head;
            _head = _tail = nullptr;
            --this->_length;
            return;
        }
        else {
            Node *t = _get_node(this->_length - 2);
            t->_next = nullptr;
            delete _tail;
            _tail = t;
            --this->_length;
        }
    }
};
#endif

    #include <iostream>
#include <cassert>
#include <cstddef>

template<typename T>
struct Node {
    T _data;
    Node* _next = nullptr;  
    Node* _prev = nullptr;  
    
    Node(T d) : _data(d) {}
};

template<typename T>
class DoublyLinkedList {
private:
    Node<T> *_head; 
    Node<T> *_tail; 
    size_t _length;

public:
    DoublyLinkedList() : _head(nullptr), _tail(nullptr), _length(0) {}

    ~DoublyLinkedList() {
        clear();
    }

    void clear() {
        Node<T> *curr = _head;
        while (curr != nullptr) {
            Node<T> *next = curr->_next;
            delete curr;
            curr = next;
        }
        _head = _tail = nullptr;
        _length = 0;
    }

    size_t size() const { return _length; }

    bool is_empty() const { return _length == 0; }

    void push_back(T d) {
        Node<T> *new_node = new Node<T>(d);
        if (_tail == nullptr) {
            _head = _tail = new_node;
        } 
        else {
            new_node->_prev = _tail;
            _tail->_next = new_node;
            _tail = new_node;
        }
        ++_length;
    }

    void push_front(T d) {
        Node<T> *new_node = new Node<T>(d);
        if (_head == nullptr) {
            _head = _tail = new_node;
        } else {
            new_node->_next = _head;
            _head->_prev = new_node;
            _head = new_node;
        }
        ++_length;
    }

    void insert_before(Node<T> *p, T d) {
        if (p == nullptr) return;
        
        Node<T> *new_node = new Node<T>(d);
        
        // 1. 先把新节点的两个指针接好
        new_node->_next = p;
        new_node->_prev = p->_prev;
        
        // 2. 处理 p 原来的前驱节点（如果有的话）
        if (p->_prev != nullptr) {
            p->_prev->_next = new_node;
        } else {
            // 如果 p 是头节点，说明新节点成了新的头节点
            _head = new_node;
        }
        
        // 3. 处理 p 的前驱指针
        p->_prev = new_node;
        
        ++_length;
    }

    void remove(Node<T> *p) {
        if (p == nullptr) return;

        // 1. 处理前驱
        if (p->_prev != nullptr) {
            p->_prev->_next = p->_next;
        } else {
            _head = p->_next; // 删除的是头节点
        }

        // 2. 处理后继
        if (p->_next != nullptr) {
            p->_next->_prev = p->_prev;
        } else {
            _tail = p->_prev; // 删除的是尾节点
        }

        // 3. 释放自身
        delete p;
        --_length;
    }

    // 正向打印
    void print_forward() {
        Node<T> *curr = _head;
        std::cout << "正向: [";
        while (curr != nullptr) {
            std::cout << curr->_data;
            if (curr->_next != nullptr) std::cout << ", ";
            curr = curr->_next;
        }
        std::cout << "]\n";
    }

    // 反向打印
    void print_backward() {
        Node<T> *curr = _tail;
        std::cout << "反向: [";
        while (curr != nullptr) {
            std::cout << curr->_data;
            if (curr->_prev != nullptr) std::cout << ", ";
            curr = curr->_prev;
        }
        std::cout << "]\n";
    }
    
    Node<T> *get_node(size_t idx) {
        assert(idx < _length);
        Node<T>* curr = _head;
        for (size_t i = 0; i < idx; ++i) curr = curr->_next;
        return curr;
    }
};

int main() {
    DoublyLinkedList<std::string> list;
    
    list.push_back("A");
    list.push_back("B");
    list.push_back("C");
    list.print_forward();  
    list.print_backward(); 

    std::cout << "--- 测试 insert_before ---\n";
    Node<std::string>* p = list.get_node(1);
    list.insert_before(p, "X"); 
    list.print_forward(); 
    list.print_backward();

    std::cout << "--- 测试 remove ---\n";
    p = list.get_node(2);
    list.remove(p);
    list.print_forward();
    list.print_backward();

    return 0;
}
